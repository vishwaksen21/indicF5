huggingface-hub<1.0

# Upstream constraint.
numpy<=1.26.4

# Official IndicF5 package (installs as `f5_tts`), pinned to the reviewed commit.
f5_tts @ git+https://github.com/ai4bharat/IndicF5.git@13f7c4d627cc10111aea8fe9c0039462cacacdc7

# Optional local UI (scripts/app.py). Additive only: installs no changes to the pins above.
